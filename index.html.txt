    function getConfigValue(config, key) {
      return config[key] || defaultConfig[key];
    }

    function showPage(page) {
      document.querySelectorAll('.page-content').forEach(section => {
        section.classList.remove('active');
      });

      const selectedPage = document.getElementById(page + '-page');
      if (selectedPage) {
        selectedPage.classList.add('active');
      }

      document.querySelectorAll('.nav-button').forEach(button => {
        button.classList.toggle('active', button.dataset.page === page);
      });

      const app = document.getElementById('app');
      if (app) {
        app.scrollTo({ top: 0, behavior: 'smooth' });
      }
    }

    let toastTimer;

    function showMessage(message) {
      const toast = document.getElementById('toast');
      toast.textContent = message;
      toast.classList.remove('opacity-0');
      toast.classList.add('opacity-100');

      clearTimeout(toastTimer);
      toastTimer = setTimeout(() => {
        toast.classList.add('opacity-0');
        toast.classList.remove('opacity-100');
      }, 2500);
    }

    function applyConfig(config) {
      const backgroundColor = getConfigValue(config, 'background_color');
      const surfaceColor = getConfigValue(config, 'surface_color');
      const textColor = getConfigValue(config, 'text_color');
      const primaryActionColor = getConfigValue(config, 'primary_action_color');
      const secondaryActionColor = getConfigValue(config, 'secondary_action_color');
      const fontFamily = getConfigValue(config, 'font_family');
      const fontSize = Number(getConfigValue(config, 'font_size')) || 16;
      const fontStack = `${fontFamily}, Nunito, sans-serif`;

      `;

      const badge = document.getElementById('hero-badge');
      if (badge && badge.lastChild) {
        badge.lastChild.textContent = " " + getConfigValue(config, 'hero_badge');

      if (siteTitle) siteTitle.textContent = getConfigValue(config, 'site_title');

      const heroSubtitle = document.getElementById('hero-subtitle');
      if (heroSubtitle) heroSubtitle.textContent = getConfigValue(config, 'hero_subtitle');

      if (primaryBtn) primaryBtn.textContent = getConfigValue(config, 'primary_button');
      const secondaryBtn = document.getElementById('secondary-button');
      if (secondaryBtn) secondaryBtn.textContent = getConfigValue(config, 'secondary_button');
      const aboutTitle = document.getElementById('about-title');
      if (aboutTitle) aboutTitle.textContent = getConfigValue(config, 'about_title');
      const aboutText = document.getElementById('about-text');
      if (aboutText) aboutText.textContent = getConfigValue(config, 'about_text');
      const footerText = document.getElementById('footer-text');
      if (footerText) footerText.textContent = getConfigValue(config, 'footer_text');

        el.style.color = textColor;
      });

        el.style.fontFamily = fontStack;
      });

      document.getElementById('hero-subtitle').style.fontSize = `${fontSize * 1.25}px`;

        el.style.fontSize = `${fontSize * 1.45}px`;
      });
      document.querySelectorAll('p').forEach(el => {
        el.style.fontSize = `${fontSize}px`;
      });

        el.style.fontSize = `${fontSize * 0.95}px`;
      });

        el.style.backgroundColor = surfaceColor + "CC";
      });

        el.style.color = primaryActionColor;
      });
